func _initialize() -> void:
	img = Image.create(CW * 7, CH, false, Image.FORMAT_RGBA8)
	img.fill(Color(0, 0, 0, 0))
	_reyes(0)
	_martelino(1)
	_obelisk(2)
	_mijares(3)
	_castillo(4)
	_rizal(5)
	_bonifacio(6)
	if errors > 0:
		print("ERRORS: ", errors)
	var out := "res://assets/pastrana/pastrana_statues.png"
	img.save_png(out)
	# Big preview for review.
	var big := img.duplicate()
	big.resize(img.get_width() * 6, img.get_height() * 6, Image.INTERPOLATE_NEAREST)
	var bg := Image.create(big.get_width(), big.get_height(), false, Image.FORMAT_RGBA8)
	bg.fill(Color("#49cf98"))
	bg.blend_rect(big, Rect2i(Vector2i.ZERO, big.get_size()), Vector2i.ZERO)
	bg.save_png(OS.get_cmdline_user_args()[0] if OS.get_cmdline_user_args().size() > 0 and OS.get_cmdline_user_args()[0].ends_with(".png") else "user://statues_preview.png")
	print("done")
	quit()

# ---------- primitives ----------
func px(x: int, y: int, c: String) -> void:
	if x < 0 or y < 0 or x >= img.get_width() or y >= img.get_height():
		return
	img.set_pixel(x, y, Color(c))

func rect(x: int, y: int, w: int, h: int, c: String) -> void:
	for yy in h:
		for xx in w:
			px(x + xx, y + yy, c)

## Outlined, shaded box: light top row and left column, dark right column.
func box(x: int, y: int, w: int, h: int, p: Dictionary) -> void:
	rect(x, y, w, h, p["o"])
	rect(x + 1, y + 1, w - 2, h - 2, p["m"])
	rect(x + 1, y + 1, w - 2, 1, p["h"])
	rect(x + 1, y + 2, 1, h - 3, p["l"])
	rect(x + w - 2, y + 2, 1, h - 3, p["d"])

## A box whose bottom row is in shadow (ledges, plinths).
func ledge(x: int, y: int, w: int, h: int, p: Dictionary) -> void:
	box(x, y, w, h, p)
	if h > 3:
		rect(x + 1, y + h - 2, w - 2, 1, p["d"])

func grid(x: int, y: int, rows: Array, p: Dictionary, name: String) -> void:
	for r in rows.size():
		var row: String = rows[r]
		if row.length() != 16:
			print("%s row %d is %d wide: %s" % [name, r, row.length(), row])
			errors += 1
		for c in row.length():
			var ch := row[c]
			if ch != ".":
				px(x + c, y + r, p[ch])

## Plaque with a frame and text lines.
func plaque(x: int, y: int, w: int, h: int, face: String, frame: String, text: String, title: String = "") -> void:
	rect(x, y, w, h, frame)
	rect(x + 1, y + 1, w - 2, h - 2, face)
	var ty := y + 2
	if title != "":
		rect(x + 2, ty, w - 4, 1, title)
		ty += 2
	while ty < y + h - 2:
		var lw := w - 4 - (1 if (ty / 2) % 3 == 2 else 0)
		rect(x + 2, ty, lw, 1, text)
		ty += 2

func _reyes(cell: int) -> void:
	var x0 := cell * CW
	# tiled two-step base
	ledge(x0 + 2, 89, 28, 7, P_TILE)
	for i in range(4, 28, 4):
		px(x0 + 2 + i, 91, P_TILE["d"])
		px(x0 + 2 + i, 92, P_TILE["d"])
	ledge(x0 + 5, 84, 22, 6, P_TILE)
	# black granite pedestal with tan corner strips
	box(x0 + 8, 48, 16, 37, P_GRANITE)
	rect(x0 + 8, 48, 2, 37, P_TILE["o"]); rect(x0 + 9, 49, 1, 35, P_TILE["l"])
	rect(x0 + 22, 48, 2, 37, P_TILE["o"]); rect(x0 + 22, 49, 1, 35, P_TILE["d"])
	ledge(x0 + 7, 45, 18, 4, P_CONCRETE)
	ledge(x0 + 7, 71, 18, 3, P_CONCRETE)
	# NHCP cast-iron marker: seal and gold lines
	plaque(x0 + 11, 52, 10, 16, "#2f2d36", "#c9a55a", "#8a6f30")
	px(x0 + 15, 53, "#e6cd8a"); px(x0 + 16, 53, "#e6cd8a")
	# bronze archbishop
	grid(x0 + 8, 16, ARCHBISHOP, P_BRONZE, "archbishop")

func _obelisk(cell: int) -> void:
	var x0 := cell * CW
	ledge(x0 + 3, 89, 26, 7, P_MARBLE)
	ledge(x0 + 6, 84, 20, 6, P_MARBLE)
	box(x0 + 8, 64, 16, 21, P_MARBLE)
	ledge(x0 + 7, 61, 18, 4, P_MARBLE)
	plaque(x0 + 11, 67, 10, 14, "#3e3d40", "#8d989e", "#adc1cc")
	# tapered shaft: 12 wide at the bottom, 6 at the top
	var top := 10
	var bottom := 61
	for y in range(top, bottom):
		var t := float(y - top) / float(bottom - top)
		var width := 2 * int(round(lerp(3.0, 6.0, t)))
		var left := x0 + 16 - width / 2
		var right := left + width - 1
		for x in range(left, right + 1):
			var c: String = P_MARBLE["m"]
			if x == left or x == right:
				c = OUT
			elif x == left + 1:
				c = P_MARBLE["h"]
			elif x == left + 2:
				c = P_MARBLE["l"]
			elif x >= right - 2:
				c = P_MARBLE["d"]
			px(x, y, c)
	# pyramidion
	for i in 6:
		var y := top - 6 + i
		var half := i / 2 + 1
		for x in range(x0 + 16 - half, x0 + 16 + half):
			var c: String = P_MARBLE["l"] if x < x0 + 16 else P_MARBLE["d"]
			if x == x0 + 16 - half or x == x0 + 16 + half - 1:
				c = OUT
			px(x, y, c)
	px(x0 + 15, top - 6, OUT); px(x0 + 16, top - 6, OUT)
	# a few veins
	for v in [[14, 20], [15, 21], [15, 22], [17, 35], [18, 36], [13, 47], [14, 48], [18, 55]]:
		px(x0 + v[0], v[1], P_MARBLE["d"])

## A boulder: outlined disc, lit top-left, shaded bottom-right.
func boulder(cx: int, cy: int, rx: int, ry: int, p: Dictionary) -> void:
	for y in range(cy - ry, cy + ry + 1):
		for x in range(cx - rx, cx + rx + 1):
			var nx := float(x - cx) / (rx + 0.5)
			var ny := float(y - cy) / (ry + 0.5)
			var d := nx * nx + ny * ny
			if d > 1.0:
				continue
			var c: String = p["m"]
			if d > 0.62:
				c = p["o"]
			elif nx + ny < -0.55:
				c = p["h"]
			elif nx + ny < -0.15:
				c = p["l"]
			elif nx + ny > 0.5:
				c = p["d"]
			px(x, y, c)
