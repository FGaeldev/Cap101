func _initialize() -> void:
	var dir := "res://assets/ui/map_icons"
	DirAccess.make_dir_recursive_absolute(ProjectSettings.globalize_path(dir))
	var ok := true
	var icons := _icons()
	for id in icons:
		var rows: Array = icons[id]
		if rows.size() > W:
			push_error("%s: %d rows" % [id, rows.size()]); ok = false; continue
		var img := Image.create(W, W, false, Image.FORMAT_RGBA8)
		img.fill(Color(0, 0, 0, 0))
		var top := W - rows.size()
		for ry in rows.size():
			var row: String = rows[ry]
			if row.length() > W:
				push_error("%s row %d: length %d" % [id, ry, row.length()]); ok = false; continue
			if row.length() % 2 == 0:
				push_error("%s row %d: even length %d can't centre" % [id, ry, row.length()]); ok = false
			var left := (W - row.length()) / 2
			for i in row.length():
				var ch := row[i]
				var x := left + i
				var y := top + ry
				if ch == "#":
					ch = "k" if ((x + y) % 4 == 0 or (x - y + 100) % 4 == 0) else "m"
				if ch == ".":
					continue
				if not PALETTE.has(ch):
					push_error("%s row %d: unknown '%s'" % [id, ry, ch]); ok = false; continue
				img.set_pixel(x, y, PALETTE[ch])
		img.save_png("%s/%s.png" % [dir, id])
	print("done ok=", ok)
	quit(0 if ok else 1)
