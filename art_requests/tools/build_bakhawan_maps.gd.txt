var _step := 0
var _sheet: Texture2D

var _ts: TileSet
var rng := RandomNumberGenerator.new()

func _initialize() -> void:
	auto_accept_quit = false

func _process(_d: float) -> bool:
	_step += 1
	if _step == 2:
		rng.seed = 1990
		_sheet = load(SHEET)

		_ts = _build_tileset()
		_save(_build_indoors(), INDOOR_PATH)
		_save(_build_trail(), TRAIL_PATH)
		_plaza_trees()
		quit()
	return false

func _save(root: Node, path: String) -> void:
	var packed := PackedScene.new()
	var err := packed.pack(root)
	if err != OK:
		push_error("pack failed for %s: %d" % [path, err])
		return
	err = ResourceSaver.save(packed, path)
	print("saved ", path, " -> ", err)

## Plaza (hand-made scene, edited as text so nothing else in it changes): the

## node in the file. Replaces any earlier "Mangroves" block (tree instances or layer).

func _plaza_trees() -> void:
	var layer := TileMapLayer.new()
	layer.tile_set = _ts
	var kinds: Array = TREES.keys()
	for t in PLAZA_TREES:
		layer.set_cell(Vector2i(t[0], t[1]), 0, kinds[t[2]])
	var data := Marshalls.raw_to_base64(layer.tile_map_data)
	layer.free()
	var file := ProjectSettings.globalize_path(PLAZA_PATH)
	var s := FileAccess.get_file_as_string(file)
	var cut := s.find("\n[node name=\"Mangroves\"")
	if cut >= 0:
		s = s.substr(0, cut) + "\n"
	s = s.replace("[ext_resource type=\"PackedScene\" uid=\"uid://bvq4cmi3t1v0h\" path=\"res://scenes/world/bakhawan/bakhawan_tree.tscn\" id=\"4_tree\"]\n", "")
	var ts_line := "[ext_resource type=\"TileSet\" path=\"%s\" id=\"5_bakhawan\"]" % TILESET_PATH
	if not s.contains(ts_line):
		var at := s.find("\n\n[node ")
		s = s.substr(0, at) + "\n" + ts_line + s.substr(at)
	s = s.rstrip("\n") + "\n\n[node name=\"Mangroves\" type=\"TileMapLayer\" parent=\".\"]\ny_sort_enabled = true\ntile_map_data = PackedByteArray(\"%s\")\ntile_set = ExtResource(\"5_bakhawan\")\n" % data
	var out := FileAccess.open(file, FileAccess.WRITE)
	out.store_string(s)
	out.close()
	print("plaza trees ", PLAZA_TREES.size())

# --- tileset ---------------------------------------------------------------------

func _build_tileset() -> TileSet:
	var ts := TileSet.new()
	ts.tile_size = Vector2i(16, 16)
	ts.add_physics_layer()
	ts.set_physics_layer_collision_layer(0, 1)
	var src := TileSetAtlasSource.new()
	src.texture = _sheet
	src.texture_region_size = Vector2i(16, 16)
	ts.add_source(src, 0)
	var img: Image = _sheet.get_image()
	if not img.get_region(Rect2i(BLOCK * 16, Vector2i(16, 16))).is_invisible():
		push_error("BLOCK cell isn't empty in the sheet")
	_tile(src, BLOCK, true)
	_tile(src, WATER, false, 4, 0.35)
	_tile(src, SHORE_TOP, false, 2, 0.6)
	_tile(src, SHORE_BOTTOM, false, 2, 0.6)
	for c in SAND + MUD + ROOTS + SEEDLINGS + [PLANK_LIGHT, PLANK_DARK]:
		_tile(src, c, false)
	for key in WALK:
		for d in [Vector2i(0, 0), Vector2i(1, 0), Vector2i(0, 1), Vector2i(1, 1)]:
			_tile(src, WALK[key] + d, false)
	for c: Vector2i in TREES:
		var cells: Vector2i = TREES[c]
		src.create_tile(c, cells)
		# Drawn centred on the cell: lift it so the roots end at the cell's bottom edge.
		src.get_tile_data(c, 0).texture_origin = Vector2i(0, cells.y * 8 - 8)
	var grass := TileSetAtlasSource.new()
	grass.texture = load(OUTDOOR_SHEET)
	grass.texture_region_size = Vector2i(16, 16)
	ts.add_source(grass, 1)
	grass.create_tile(GRASS)
	var err := ResourceSaver.save(ts, TILESET_PATH)
	print("saved ", TILESET_PATH, " -> ", err)
	return load(TILESET_PATH)

func _tile(src: TileSetAtlasSource, c: Vector2i, solid: bool, frames: int = 1, seconds: float = 0.3) -> void:
	src.create_tile(c)
	if frames > 1:
		src.set_tile_animation_columns(c, frames)
		src.set_tile_animation_frames_count(c, frames)
		for i in frames:
			src.set_tile_animation_frame_duration(c, i, seconds)
	if solid:
		var data := src.get_tile_data(c, 0)
		data.add_collision_polygon(0)
		data.set_collision_polygon_points(0, 0, PackedVector2Array([Vector2(-8, -8), Vector2(8, -8), Vector2(8, 8), Vector2(-8, 8)]))

# --- helpers ---------------------------------------------------------------------

func _layer(root: Node, layer_name: String, ts: TileSet, z: int, ysort: bool = false) -> TileMapLayer:
	var l := TileMapLayer.new()
	l.name = layer_name
	l.tile_set = ts
	l.z_index = z
	l.y_sort_enabled = ysort
	root.add_child(l)
	l.owner = root
	return l

func _node(parent: Node, root: Node, n: Node, node_name: String) -> Node:
	n.name = node_name
	parent.add_child(n)
	n.owner = root
	return n

## Edge ring (edges.tres 9-slice) around cells x0..x1, y0..y1 inclusive.
func _edge_ring(l: TileMapLayer, x0: int, y0: int, x1: int, y1: int) -> void:
	for x in range(x0, x1 + 1):
		l.set_cell(Vector2i(x, y0), 0, Vector2i(0 if x == x0 else (2 if x == x1 else 1), 0))
		l.set_cell(Vector2i(x, y1), 0, Vector2i(0 if x == x0 else (2 if x == x1 else 1), 2))
	for y in range(y0 + 1, y1):
		l.set_cell(Vector2i(x0, y), 0, Vector2i(0, 1))
		l.set_cell(Vector2i(x1, y), 0, Vector2i(2, 1))

## A sprite prop standing on footprint cells (fx, fy, fw x fh), art centred on it
## and sitting on its bottom edge. "solid" = cells, relative to the footprint, that collide.
func _prop(parent: Node, root: Node, prop_name: String, art: String, fx: int, fy: int, fw: int, fh: int,
		solid: Rect2i = Rect2i(), tex: Texture2D = null, region: Rect2i = Rect2i(), z: int = 0, lift: int = 0) -> Node2D:
	var holder := Node2D.new()
	_node(parent, root, holder, prop_name)
	holder.position = Vector2(fx * 16, (fy + fh) * 16)
	var r: Rect2i = region if region.has_area() else ART[art]
	var s := Sprite2D.new()
	_node(holder, root, s, "Sprite2D")
	s.texture = tex if tex != null else _sheet
	s.region_enabled = true
	s.region_rect = Rect2(r)
	s.centered = false
	s.position = Vector2(int((fw * 16 - r.size.x) / 2.0), -r.size.y - lift)
	s.z_index = z
	if solid.has_area():
		var body := StaticBody2D.new()
		_node(holder, root, body, "Body")
		var shape := CollisionShape2D.new()
		_node(body, root, shape, "Shape")
		var rect := RectangleShape2D.new()
		rect.size = Vector2(solid.size * 16)
		shape.shape = rect
		shape.position = Vector2(solid.position.x * 16 + solid.size.x * 8, (solid.position.y - fh) * 16 + solid.size.y * 8)
	return holder

func _marker(parent: Node, root: Node, marker_name: String, cell: Vector2i) -> void:
	var m := Marker2D.new()
	_node(parent, root, m, marker_name)
	m.position = Vector2(cell.x * 16, (cell.y + 1) * 16)

## Door on cells x..x+1 of row "row", walked through in "dir" (comp_warp TriggerDirection).
func _warp(parent: Node, root: Node, warp_name: String, x: int, row: int, dir: int, target: String, spawn: String, closed: String = "") -> void:
	var w: Node2D = (load(WARP_SCENE) as PackedScene).instantiate()
	_node(parent, root, w, warp_name)
	w.position = Vector2((x + 1) * 16, row * 16 + 8)
	w.set("trigger_direction", dir)
	w.set("span_tiles_override", 2)
	if not target.is_empty():
		w.set("target_scene", target)
		w.set("target_spawn_id", spawn)
	if not closed.is_empty():
		w.set("closed_message", closed)

## A tree on base cell "base" (under its trunk) covers cells.x columns around it and
## cells.y rows up to it. True if all of that, plus a 1-cell margin, is clear of
## walkable cells and inside the map.
func _tree_fits(walkable: Dictionary, base: Vector2i, cells: Vector2i) -> bool:
	var x0: int = base.x - cells.x / 2
	for yy in range(base.y - cells.y, base.y + 2):
		for xx in range(x0 - 1, x0 + cells.x + 1):
			if xx < 0 or xx >= W or walkable.has(Vector2i(xx, yy)):
				return false
	return true

func _pick(list: Array, c: Vector2i):
	return list[abs(c.x * 7 + c.y * 13) % list.size()]

## Boardwalk in 2x2 cells (cell (cx, cy) covers tiles 2cx..2cx+1, 2cy..2cy+1).
func _walk_cells() -> Dictionary:
	var cells := {}
	var path := [Vector2i(9, 27), Vector2i(9, 26), Vector2i(9, 25), Vector2i(9, 24), Vector2i(9, 23),
		Vector2i(8, 23), Vector2i(7, 23), Vector2i(6, 23), Vector2i(5, 23), Vector2i(4, 23),
		Vector2i(4, 22), Vector2i(4, 21), Vector2i(4, 20), Vector2i(4, 19), Vector2i(4, 18),
		Vector2i(5, 18), Vector2i(5, 17), Vector2i(6, 17), Vector2i(6, 16), Vector2i(7, 16), Vector2i(7, 15),
		Vector2i(8, 15), Vector2i(8, 14), Vector2i(9, 14), Vector2i(10, 14)]
	for y in range(3, 14):
		path.append(Vector2i(10, y))
	for x in range(6, 17):
		path.append(Vector2i(x, 3))
	for c in path:
		cells[c] = true
	for c in [Vector2i(5, 19), Vector2i(6, 19), Vector2i(5, 20), Vector2i(6, 20)]:  # roots platform
		cells[c] = true
	return cells

func _piece(cells: Dictionary, c: Vector2i) -> Vector2i:
	var key := ""
	if cells.has(c + Vector2i.UP): key += "N"
	if cells.has(c + Vector2i.RIGHT): key += "E"
	if cells.has(c + Vector2i.DOWN): key += "S"
	if cells.has(c + Vector2i.LEFT): key += "W"
	if c.y == 27:
		key = "NS"  # first piece, off the land
	if WALK.has(key):
		return WALK[key]
	if key == "N":
		return WALK["NS"]
	return WALK["EW"]  # T-joints we have no art for, the platform, the deck ends

func _build_trail() -> Node2D:
	var root := Node2D.new()
	root.name = "BakhawanTrail"
	root.y_sort_enabled = true
	root.set_script(load("res://scripts/game_scenes/bakhawan_trail.gd"))
	# Ground and water come from outdoors.tres (the team's tiles in pastrana_outdoors.png).
	var outdoors: TileSet = load(OUTDOORS_TS)
	var floor_l := _layer(root, "Floor", outdoors, -3)
	var tints: Array[TileMapLayer] = []
	for i in WATER_TINTS.size():
		var tint := _layer(root, "Water%d" % i, outdoors, -3)
		tint.self_modulate = WATER_TINTS[i]
		tints.append(tint)
	var water_l := _layer(root, "Water", _ts, -3)
	var walk_l := _layer(root, "Boardwalk", _ts, -2)
	var decals := _layer(root, "FloorDecals", _ts, -2)
	var edge := _layer(root, "Edge", load(EDGES_TS), 0)

	var walkable := {}
	var cells := _walk_cells()
	for c: Vector2i in cells:
		var piece := _piece(cells, c)
		for d in [Vector2i(0, 0), Vector2i(1, 0), Vector2i(0, 1), Vector2i(1, 1)]:
			walk_l.set_cell(c * 2 + d, 0, piece + d)
			walkable[c * 2 + d] = true
	for y in H:
		for x in W:
			var c := Vector2i(x, y)
			if y >= 56:
				# Land by the gate: a muddy bank along the water, grass behind it.
				floor_l.set_cell(c, 0, OD_GRASS_MUD_TOP if y == 56 else OD_GRASS)
				walkable[c] = true
			elif x >= 14 and x <= 31 and y >= 20 and y <= 27:
				# Rest island: grass with a mud rim (the grass-in-mud 9-slice).
				var col := 0 if x == 14 else (2 if x == 31 else 1)
				var row := 0 if y == 20 else (2 if y == 27 else 1)

				walkable[c] = true
			else:
				# Water: greenish among the mangroves, bluer toward the open sea,
				# dithered between tint bands so the change is gradual.
				var t: float = clampf(float(55 - y) / 55.0, 0.0, 1.0)
				var dither: float = BAYER[y % 4][x % 4] / 16.0 - 0.47
				var band: int = clampi(int(floor(t * (WATER_TINTS.size() - 1) + 0.5 + dither)), 0, WATER_TINTS.size() - 1)
				tints[band].set_cell(c, 0, OD_WATER)
